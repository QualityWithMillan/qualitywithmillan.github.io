<script>
  (function () {
    try {
      var currentPath = window.location.pathname;
      var search = window.location.search || '';
      var hash = window.location.hash || '';

      if (!currentPath || currentPath === '/' || currentPath === '/404.html') {
        return;
      }

      var attempts = [];
      try {
        attempts = JSON.parse(sessionStorage.getItem('404_redirect_attempts') || '[]');
      } catch (err) {
        attempts = [];
      }

      if (!Array.isArray(attempts) || attempts.length > 6) {
        attempts = [];
      }

      if (!attempts.includes(currentPath)) {
        attempts.push(currentPath);
        sessionStorage.setItem('404_redirect_attempts', JSON.stringify(attempts));

        // 1. If path ends with trailing slash, try removing it (e.g., /path/ -> /path)
        if (currentPath.length > 1 && currentPath.endsWith('/')) {
          var trimmedPath = currentPath.replace(/\/+$/, '');
          if (trimmedPath && !attempts.includes(trimmedPath)) {
            window.location.replace(trimmedPath + search + hash);
            return;
          }
        }
        // 2. If path has no trailing slash and no file extension, try adding trailing slash (e.g., /path -> /path/)
        else if (!currentPath.endsWith('/') && !currentPath.split('/').pop().includes('.')) {
          var slashPath = currentPath + '/';
          if (!attempts.includes(slashPath)) {
            window.location.replace(slashPath + search + hash);
            return;
          }
        }
      }
    } catch (e) {
      console.warn('404 redirect handler:', e);
    }
  })();
</script>
