<script>
  // --- Chatbot logic ---

  // Get references
  const chatBtn = document.getElementById('chat-wave-btn');
  const chatWindow = document.getElementById('chat-window');
  const chatMessages = document.getElementById('chat-messages');
  const chatForm = document.getElementById('chat-form');
  const chatInput = document.getElementById('chat-input');
  const chatSendBtn = document.getElementById('chat-send-btn');
  const chatClearBtn = document.getElementById('chat-clear-btn');

  // Toggle chat window visibility
  chatBtn.addEventListener('click', () => {
    if (chatWindow.style.display === 'flex') {
      chatWindow.style.display = 'none';
    } else {
      chatWindow.style.display = 'flex';
      chatInput.focus();
    }
  });

  // Enable send button only if input not empty
  chatInput.addEventListener('input', () => {
    chatSendBtn.disabled = chatInput.value.trim() === '';
  });

  // Extract blog posts content on load
  const blogPosts = Array.from(document.querySelectorAll('.blog-post')).map(post => {
    return {
      title: post.querySelector('h2')?.innerText || '',
      text: post.innerText || '',
      url: '#' + post.id
    };
  });

  // Simple function to find best matching blog snippet
  function findBestMatch(question) {
    let maxScore = 0;
    let bestMatch = null;
    const words = question.toLowerCase().split(/\W+/).filter(Boolean);

    blogPosts.forEach(post => {
      const postText = post.text.toLowerCase();
      let score = words.reduce((acc, w) => acc + (postText.includes(w) ? 1 : 0), 0);
      if (score > maxScore) {
        maxScore = score;
        bestMatch = post;
      }
    });

    return bestMatch;
  }

  // Bot response builder
  function chatbotRespond(question) {
    if (!question.trim()) {
      return "Please ask a question!";
    }

    const match = findBestMatch(question);
    if (match && match.title) {
      // Extract short excerpt (first 150 chars) from the post text for preview
      let excerpt = match.text.replace(/\s+/g, ' ').trim().slice(0, 150);
      if (match.text.length > 150) excerpt += "...";

      return `I found this post that might help you: <br><br> <strong>${match.title}</strong>.<br><br>${excerpt} <a href="${match.url}" onclick="event.preventDefault(); scrollToPost('${match.url.substring(1)}');">Read more</a>`;
    } else {
      return "Sorry, I couldn't find an answer. Maybe try once more please?";
    }
  }

  // Scroll smoothly to a blog post
  function scrollToPost(postId) {
    chatWindow.style.display = 'none'; // Close chat window
    const postElem = document.getElementById(postId);
    if (postElem) {
      postElem.scrollIntoView({ behavior: 'smooth' });
      postElem.style.backgroundColor = '#ffffcc';
      setTimeout(() => postElem.style.backgroundColor = '', 2000);
    }
  }

  // Append message to chat window
  function appendMessage(text, sender = 'bot') {
    const msgDiv = document.createElement('div');
    msgDiv.classList.add('chat-message', sender);
    msgDiv.innerHTML = text;
    chatMessages.appendChild(msgDiv);
    chatMessages.scrollTop = chatMessages.scrollHeight;
  }

  // Handle form submission
  chatForm.addEventListener('submit', e => {
    e.preventDefault();
    const question = chatInput.value.trim();
    if (!question) return;

    // Show user's message
    appendMessage(question, 'user');
    chatInput.value = '';
    chatSendBtn.disabled = true;

    // Bot reply with slight delay to mimic thinking
    setTimeout(() => {
      const reply = chatbotRespond(question);
      appendMessage(reply, 'bot');
    }, 600);
  });

  // Clear chat messages
  chatClearBtn.addEventListener('click', () => {
    chatMessages.innerHTML = '';
    chatInput.value = '';
    chatSendBtn.disabled = true;
    chatInput.focus();
  });

  // Accessibility: open chat with keyboard (Enter/Space)
  chatBtn.addEventListener('keydown', e => {
    if (e.key === 'Enter' || e.key === ' ') {
      e.preventDefault();
      chatBtn.click();
    }
  });

</script>
